/*

  Most Arduinos have an on-board LED you can control. On the UNO, MEGA and ZERO
  it is attached to digital pin 13, on MKR1000 on pin 6. LED_BUILTIN is set to
  the correct LED pin independent of which board is used.
  If you want to know what pin the on-board LED is connected to on your Arduino
  model, check the Technical Specs of your board at:

#define MIDI_MODE_DISPLAY 1 << 0
#define MIDI_MODE_SERIAL 1 << 1
#define MIDI_MODE_USB 1 << 2

//#define MIDI_MODE MIDI_MODE_DISPLAY | MIDI_MODE_SERIAL | MIDI_MODE_USB
#define MIDI_MODE MIDI_MODE_DISPLAY

// enum representing each note
enum class Note {
  C = 0,
  C_SHARP = 1,
  D = 2,
  D_SHARP = 3,
  E = 4,
  F = 5,
  F_SHARP = 6,
  G = 7,
  G_SHARP = 8,
  A = 9,
  A_SHARP = 10,
  B = 11
};

enum class MidiCommand {
  NOTE_OFF = 0x80,
  NOTE_ON = 0x90,
  POLY_PRESSURE = 0xA0,
  CONTROL_CHANGE = 0xB0,
  PROGRAM_CHANGE = 0xC0,
  CHANNEL_PRESSURE = 0xD0,
  PITCH_BEND = 0xE0
};

void noteToString(Note note, char *str) {
  switch (note) {
    case Note::C:
      strncpy(str, "C", 2);
      break;
    case Note::C_SHARP:
      strncpy(str, "C#", 3);
      break;
    case Note::D:
      strncpy(str, "D", 2);
      break;
    case Note::D_SHARP:
      strncpy(str, "D#", 3);
      break;
    case Note::E:
      strncpy(str, "E", 2);
      break;
    case Note::F:
      strncpy(str, "F", 2);
      break;
    case Note::F_SHARP:
      strncpy(str, "F#", 3);
      break;
    case Note::G:
      strncpy(str, "G", 2);
      break;
    case Note::G_SHARP:
      strncpy(str, "G#", 3);
      break;
    case Note::A:
      strncpy(str, "A", 2);
      break;
    case Note::A_SHARP:
      strncpy(str, "A#", 3);
      break;
    case Note::B:
      strncpy(str, "B", 2);
      break;
  }
}

void commandToString(MidiCommand command, char *str) {
  switch (command) {
    case MidiCommand::NOTE_OFF:
      strncpy(str, "Note Off", 9);
      break;
    case MidiCommand::NOTE_ON:
      strncpy(str, "Note On", 8);
      break;
    case MidiCommand::POLY_PRESSURE:
      strncpy(str, "Poly Pressure", 13);
      break;
    case MidiCommand::CONTROL_CHANGE:
      strncpy(str, "Control Change", 14);
      break;
    case MidiCommand::PROGRAM_CHANGE:
      strncpy(str, "Program Change", 14);
      break;
    case MidiCommand::CHANNEL_PRESSURE:
      strncpy(str, "Channel Pressure", 16);
      break;
    case MidiCommand::PITCH_BEND:
      strncpy(str, "Pitch Bend", 10);
      break;
  }
}

void writeMidiNote(MidiCommand command, Note note, int octave, int velocity) {
  if (MIDI_MODE & MIDI_MODE_DISPLAY) {
    char noteStr[3], commandString[32];
    commandToString(command, commandString);
    noteToString(note, noteStr);

    Serial.print(commandString);
    Serial.print(": ");
    if (command == MidiCommand::NOTE_ON || command == MidiCommand::NOTE_OFF) {
      Serial.print(noteStr);
      Serial.print("-");
      Serial.print(octave);
      Serial.print(" ");
      Serial.println(velocity);
    }
  }
}

// the setup function runs once when you press reset or power the board
void setup() {
  // initialize digital pin LED_BUILTIN as an output.
  pinMode(LED_BUILTIN, OUTPUT);
  // activate the serial monitor
  Serial.begin(115200);
}

// the loop function runs over and over again forever
void loop() {
  // print blinking on the serial monitor
  digitalWrite(LED_BUILTIN, HIGH);  // turn the LED on (HIGH is the voltage level)
  delay(1000);                      // wait for 1000 ms
  digitalWrite(LED_BUILTIN, LOW);   // turn the LED off by making the voltage LOW
  delay(1000);                      // wait for 1000 ms
  writeMidiNote(MidiCommand::NOTE_ON, Note::C, 4, 127);
  writeMidiNote(MidiCommand::NOTE_OFF, Note::C, 4, 127);
}
